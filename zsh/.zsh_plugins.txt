# Plugins de zsh — gestionados por antidote
#
# El orden IMPORTA:
#   completions -> compinit -> resto -> syntax highlighting -> history search
#
# Anadir uno: escribirlo aqui y `exec zsh`. antidote lo clona y regenera
# el fichero estatico solo. Catalogo: github.com/topics/zsh-plugins
#
# OJO con los plugins de ohmyzsh: algunos dependen de variables internas del
# framework (ZSH_CACHE_DIR). El .zshrc la define como red de seguridad, pero
# antes de anadir uno, comprueba que no duplique algo que brew ya instala
# en /opt/homebrew/share/zsh/site-functions (ahi ya estan gh, brew, mise,
# uv, rustup, bat, eza, fd, rg, zoxide, starship y ghostty).

# ---------- Completions ----------
# Completions extra para cientos de comandos que zsh no cubre de serie
zsh-users/zsh-completions kind:fpath

# Ejecuta compinit en el momento correcto y cachea el dump.
# DESPUES de las fpath, ANTES de lo que dependa de completions.
mattmc3/ez-compinit

# Tab completion con interfaz fzf: selector con busqueda difusa en vez de
# una lista plana. Es el plugin que mas se nota en el dia a dia.
Aloxaf/fzf-tab

# ---------- De Oh My Zsh, solo lo util ----------
# Compatible sin cargar el framework entero (3054 lineas que nos ahorramos)
ohmyzsh/ohmyzsh path:plugins/git       # alias: gst, gco, gcm, glog...
ohmyzsh/ohmyzsh path:plugins/macos     # ofd, pfd, cdf, showfiles...
ohmyzsh/ohmyzsh path:plugins/extract   # `extract fichero` para cualquier formato

# ---------- Experiencia ----------
# Sugiere el comando completo en gris segun escribes (Flecha derecha acepta)
zsh-users/zsh-autosuggestions

# ---------- Al final, en este orden ----------
# Mas rapido y preciso que zsh-syntax-highlighting. kind:defer lo carga
# tras dibujar el prompt, asi no retrasa el arranque.
zdharma-continuum/fast-syntax-highlighting kind:defer

# Flecha arriba busca en el historial por lo ya escrito.
# DESPUES del highlighting o se pisan los bindings.
zsh-users/zsh-history-substring-search
