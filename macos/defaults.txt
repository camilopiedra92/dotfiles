# macOS settings this machine is expected to have, one per line.
#
# Four whitespace-separated fields: domain, key, type, value. `#` starts a
# comment. The domain may be written `host:NSGlobalDomain` instead of
# `NSGlobalDomain`: macOS keeps some preferences -- trackpad and mouse among
# them -- per host rather than in the domain's usual file, and defaults.sh
# reads and writes a `host:` line through `defaults -currentHost` to reach
# it. macos/defaults.sh applies the file from install.sh and checks it from
# drift.sh, and is the only thing that reads it.
#
# Only values that differ from Apple's default are here. A default written
# down is frozen: the day Apple ships a better one this file would put the
# old one back, and nobody would know why. Same argument as node = "lts"
# rather than a number.
#
# No key here is one a Jamf profile sets on this machine. The rule is per
# key and not per domain: management is by key, and com.apple.controlcenter
# below is a domain a profile does touch (its Bluetooth and Wi-Fi menu
# items) while BatteryShowPercentage in the same domain is not. To check,
# `ls "/Library/Managed Preferences/$USER"` lists the managed domains and
# `plutil -p` on one of those plists lists its keys. Screen lock, updates,
# firewall, FileVault and Siri are policy there. A managed key written here
# would be silently overruled while reading as if it applied: `defaults`
# reads and writes only the user layer, and the managed one sits above it,
# invisible to `defaults read` and consulted first by the app. Checked on
# this machine: the managed Bluetooth key reads as "does not exist".
#
# Verified by read-back only, except the trackpad: on macOS 26.6.2 on
# 2026-09-15, `defaults.sh check` reports every key at its declared value,
# and tap-to-click and three-finger drag are also UI-verified -- toggling
# each in System Settings and diffing preference files showed both land in
# the per-host global domain below, not the two Trackpad domains every
# "macos.sh" template writes; see the Trackpad section. A key that turns out
# not to take effect is to be removed and named here, not patched with a
# second key copied from a template; those carry keys that stopped doing
# anything releases ago, and a line that reads back correctly while changing
# nothing is what this file exists to not have.

# ── Finder ──────────────────────────────────────────────────────────────────

# Extensions are how you tell a script from a directory; hiding them is for
# people who never see either.
NSGlobalDomain          AppleShowAllExtensions          bool    true
# Where you are and how much is there, at the bottom of every window.
com.apple.finder        ShowPathbar                     bool    true
com.apple.finder        ShowStatusBar                   bool    true
# List view: dates, sizes and sorting, which icons cannot show.
com.apple.finder        FXPreferredViewStyle            string  Nlsv
# Search the folder you are in, not the whole disk. SCcf = current folder.
com.apple.finder        FXDefaultSearchScope            string  SCcf
# Renaming a.txt to a.md is a rename, not a mistake worth a dialog.
com.apple.finder        FXEnableExtensionChangeWarning  bool    false
# In list view, a folder sorted between files by name is invisible until you
# scroll to where its name would put it.
com.apple.finder        _FXSortFoldersFirst             bool    true
# New windows open at home, not at Recents. PfHm = home directory.
com.apple.finder        NewWindowTarget                 string  PfHm
# .DS_Store on network and USB volumes litters every repo shared that way
# and every colleague's machine that mounts it.
com.apple.desktopservices  DSDontWriteNetworkStores     bool    true
com.apple.desktopservices  DSDontWriteUSBStores         bool    true

# ── Dock and Mission Control ────────────────────────────────────────────────

# Already the state of this machine; declared so a rebuild reproduces it.
com.apple.dock          autohide                        bool    true
com.apple.dock          minimize-to-application         bool    true
# The hidden Dock appears the moment the cursor reaches the edge, not after
# the half-second Apple adds.
com.apple.dock          autohide-delay                  float   0
# Recents in the Dock is a second, unsorted app switcher.
com.apple.dock          show-recents                    bool    false
# Spaces stay where you put them. With this on, the one you used last moves
# to the front and Ctrl+number stops meaning anything.
com.apple.dock          mru-spaces                      bool    false

# ── Keyboard and text ───────────────────────────────────────────────────────

# Repeat fast enough that holding a key in an editor is a way of moving.
# Apple's slider bottoms out at 2 / 15; these are that setting.
NSGlobalDomain          KeyRepeat                       int     2
NSGlobalDomain          InitialKeyRepeat                int     15
# Press-and-hold stays ON globally: the layout here is ABC and Spanish
# accents are typed through it. It is turned off per app, below, for the two
# where holding a key has to repeat it.
com.microsoft.VSCode    ApplePressAndHoldEnabled        bool    false
com.mitchellh.ghostty   ApplePressAndHoldEnabled        bool    false
# Every automatic substitution rewrites code and shell text pasted into a
# chat, a ticket or a form: a straight quote becomes a curly one, two dashes
# an em dash, a lowercase identifier a Capitalised one. Off everywhere; the
# apps where prose is written have their own switches.
NSGlobalDomain          NSAutomaticCapitalizationEnabled        bool    false
NSGlobalDomain          NSAutomaticPeriodSubstitutionEnabled    bool    false
NSGlobalDomain          NSAutomaticQuoteSubstitutionEnabled     bool    false
NSGlobalDomain          NSAutomaticDashSubstitutionEnabled      bool    false
NSGlobalDomain          NSAutomaticSpellingCorrectionEnabled    bool    false
NSGlobalDomain          NSAutomaticInlinePredictionEnabled      bool    false
# Tab moves between every control in a dialog, buttons included, so a save
# prompt can be answered without reaching for the trackpad.
NSGlobalDomain          AppleKeyboardUIMode             int     2

# ── Trackpad ────────────────────────────────────────────────────────────────

# com.apple.AppleMultitouchTrackpad and com.apple.driver.AppleBluetoothMulti-
# touch.trackpad Clicking/TrackpadThreeFingerDrag stood here until
# 2026-09-15: every "macos.sh" template on the internet writes them, but on
# macOS 26.6.2 nothing reads them back -- toggling tap-to-click in System
# Settings and diffing preference files showed only
# ~/Library/Preferences/ByHost/.GlobalPreferences.<uuid>.plist changing, not
# either Trackpad domain. They read back as applied and changed nothing,
# which is exactly what this file's header says to remove on sight.
#
# Tap to click. Per-host because macOS keeps trackpad and mouse settings per
# machine, which is also why the key lives in the ByHost file rather than in
# NSGlobalDomain's own plist.
host:NSGlobalDomain    com.apple.mouse.tapBehavior    int    1
# Three-finger drag. UI-verified 2026-09-15: toggled on in System Settings ->
# Accessibility -> Pointer Control -> Trackpad Options, then read back from
# ByHost .GlobalPreferences (its mtime moved). The UI writes all three keys
# below, not just the drag one -- once three fingers mean drag, the
# three-finger swipe gestures move to four fingers, so it turns the swipe
# keys off rather than leaving them pointing at a gesture that no longer
# exists. com.apple.trackpad.fourFinger{Horiz,Vert}SwipeGesture were already
# 2, Apple's own default, before and after the toggle, so they are not
# declared here.
host:NSGlobalDomain    com.apple.trackpad.threeFingerDragGesture        int   1
host:NSGlobalDomain    com.apple.trackpad.threeFingerHorizSwipeGesture  int   0
host:NSGlobalDomain    com.apple.trackpad.threeFingerVertSwipeGesture   int   0
